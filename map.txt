"use client"

import { useState, useCallback, useEffect } from "react"
import { Button } from "@/components/ui/button"
import { 
  Map, Layers, ZoomIn, ZoomOut, Locate, 
  AlertTriangle, Zap, Activity, Radio, RefreshCw,
  MapPin, Waves, ThermometerSun
} from "lucide-react"
import { GoogleMap, useJsApiLoader, Marker, Circle, InfoWindow } from "@react-google-maps/api"

const mapContainerStyle = {
  width: "100%",
  height: "100%",
}

const center = {
  lat: 28.6139,
  lng: 77.2090,
}

// Initial power grid stations
const initialStations = [
  { id: 1, name: "North Delhi Substation", lat: 28.7041, lng: 77.1025, load: 95, capacity: 500, status: "critical", type: "substation" },
  { id: 2, name: "South Delhi Substation", lat: 28.5245, lng: 77.1855, load: 72, capacity: 600, status: "warning", type: "substation" },
  { id: 3, name: "Central Delhi Grid", lat: 28.6358, lng: 77.2245, load: 45, capacity: 800, status: "normal", type: "grid" },
  { id: 4, name: "East Delhi Station", lat: 28.6280, lng: 77.2950, load: 88, capacity: 450, status: "warning", type: "substation" },
  { id: 5, name: "West Delhi Grid", lat: 28.6509, lng: 77.0688, load: 35, capacity: 550, status: "normal", type: "grid" },
  { id: 6, name: "Noida Power Hub", lat: 28.5355, lng: 77.3910, load: 82, capacity: 700, status: "warning", type: "hub" },
  { id: 7, name: "Gurgaon Substation", lat: 28.4595, lng: 77.0266, load: 55, capacity: 650, status: "normal", type: "substation" },
  { id: 8, name: "Faridabad Grid", lat: 28.4089, lng: 77.3178, load: 92, capacity: 400, status: "critical", type: "grid" },
  { id: 9, name: "Rohini Solar Farm", lat: 28.7380, lng: 77.0640, load: 28, capacity: 200, status: "normal", type: "solar" },
  { id: 10, name: "Dwarka Wind Station", lat: 28.5921, lng: 77.0460, load: 42, capacity: 180, status: "normal", type: "wind" },
]

const getLoadColor = (load: number) => {
  if (load >= 85) return "#ff4444"
  if (load >= 70) return "#ffaa00"
  if (load >= 50) return "#00ff88"
  return "#00d4ff"
}

const getStatusColor = (status: string) => {
  switch (status) {
    case "critical": return "#ff4444"
    case "warning": return "#ffaa00"
    default: return "#00ff88"
  }
}

const darkMapStyles = [
  { elementType: "geometry", stylers: [{ color: "#0a0a0f" }] },
  { elementType: "labels.text.stroke", stylers: [{ color: "#0a0a0f" }] },
  { elementType: "labels.text.fill", stylers: [{ color: "#4a4a5a" }] },
  { featureType: "administrative", elementType: "geometry.stroke", stylers: [{ color: "#1a1a2e" }] },
  { featureType: "administrative.locality", elementType: "labels.text.fill", stylers: [{ color: "#6a6a7a" }] },
  { featureType: "poi", elementType: "geometry", stylers: [{ color: "#0f0f15" }] },
  { featureType: "poi", elementType: "labels.text.fill", stylers: [{ color: "#4a4a5a" }] },
  { featureType: "road", elementType: "geometry", stylers: [{ color: "#1a1a2e" }] },
  { featureType: "road", elementType: "geometry.stroke", stylers: [{ color: "#0a0a0f" }] },
  { featureType: "road.highway", elementType: "geometry", stylers: [{ color: "#2a2a3e" }] },
  { featureType: "water", elementType: "geometry", stylers: [{ color: "#050510" }] },
  { featureType: "water", elementType: "labels.text.fill", stylers: [{ color: "#3a3a4a" }] },
]

export default function LoadMapPage() {
  const [selectedStation, setSelectedStation] = useState<typeof initialStations[0] | null>(null)
  const [map, setMap] = useState<google.maps.Map | null>(null)
  const [gridStations, setGridStations] = useState(initialStations)
  const [lastUpdate, setLastUpdate] = useState(new Date())
  const [isRefreshing, setIsRefreshing] = useState(false)

  const { isLoaded, loadError } = useJsApiLoader({
    googleMapsApiKey: process.env.NEXT_PUBLIC_GOOGLE_MAPS_API_KEY || "",
  })

  // Live updates every 3 seconds
  useEffect(() => {
    const interval = setInterval(() => {
      setGridStations(prev => prev.map(station => {
        const loadChange = (Math.random() - 0.5) * 8
        const newLoad = Math.min(100, Math.max(10, station.load + loadChange))
        const newStatus = newLoad >= 85 ? 'critical' : newLoad >= 70 ? 'warning' : 'normal'
        return { ...station, load: Math.round(newLoad), status: newStatus }
      }))
      setLastUpdate(new Date())
    }, 3000)
    return () => clearInterval(interval)
  }, [])

  const onLoad = useCallback((map: google.maps.Map) => {
    setMap(map)
  }, [])

  const onUnmount = useCallback(() => {
    setMap(null)
  }, [])

  const handleZoomIn = () => map?.setZoom((map.getZoom() || 10) + 1)
  const handleZoomOut = () => map?.setZoom((map.getZoom() || 10) - 1)
  const handleCenter = () => { map?.panTo(center); map?.setZoom(10) }
  
  const handleRefresh = () => {
    setIsRefreshing(true)
    setTimeout(() => setIsRefreshing(false), 1000)
  }

  const criticalCount = gridStations.filter(s => s.status === "critical").length
  const warningCount = gridStations.filter(s => s.status === "warning").length
  const normalCount = gridStations.filter(s => s.status === "normal").length
  const totalLoad = gridStations.reduce((acc, s) => acc + (s.load * s.capacity / 100), 0)

  // Fallback map when no API key
  const FallbackMap = () => (
    <div className="relative w-full h-full bg-[#0a0a0f] overflow-hidden">
      {/* Grid background */}
      <div className="absolute inset-0 grid-pattern opacity-30" />
      
      {/* Simulated map with stations */}
      <svg viewBox="0 0 800 600" className="w-full h-full">
        {/* Delhi NCR outline approximation */}
        <path
          d="M200,150 L350,100 L550,120 L650,200 L680,350 L620,480 L450,530 L280,500 L150,380 L180,250 Z"
          fill="none"
          stroke="rgba(0,255,136,0.2)"
          strokeWidth="2"
          strokeDasharray="5,5"
        />
        
        {/* Connection lines between stations */}
        {gridStations.map((station, i) => {
          const x = ((station.lng - 76.8) / 0.8) * 600 + 100
          const y = ((28.9 - station.lat) / 0.5) * 400 + 100
          return gridStations.slice(i + 1).map((other, j) => {
            const ox = ((other.lng - 76.8) / 0.8) * 600 + 100
            const oy = ((28.9 - other.lat) / 0.5) * 400 + 100
            const distance = Math.sqrt(Math.pow(x - ox, 2) + Math.pow(y - oy, 2))
            if (distance < 200) {
              return (
                <line
                  key={`${station.id}-${other.id}`}
                  x1={x} y1={y} x2={ox} y2={oy}
                  stroke="rgba(0,212,255,0.15)"
                  strokeWidth="1"
                />
              )
            }
            return null
          })
        })}
        
        {/* Station markers */}
        {gridStations.map((station) => {
          const x = ((station.lng - 76.8) / 0.8) * 600 + 100
          const y = ((28.9 - station.lat) / 0.5) * 400 + 100
          const color = getStatusColor(station.status)
          return (
            <g 
              key={station.id} 
              className="cursor-pointer"
              onClick={() => setSelectedStation(station)}
            >
              {/* Pulse effect */}
              <circle
                cx={x} cy={y} r={station.load / 4 + 15}
                fill={color}
                opacity={0.1}
                className="animate-pulse"
              />
              {/* Outer ring */}
              <circle
                cx={x} cy={y} r={12}
                fill="transparent"
                stroke={color}
                strokeWidth="2"
                opacity={0.5}
              />
              {/* Inner circle */}
              <circle
                cx={x} cy={y} r={8}
                fill={color}
              />
              {/* Station label */}
              <text
                x={x} y={y + 25}
                textAnchor="middle"
                fill="rgba(255,255,255,0.7)"
                fontSize="10"
                fontWeight="500"
              >
                {station.load}%
              </text>
            </g>
          )
        })}
      </svg>

      {/* Selected station info */}
      {selectedStation && (
        <div 
          className="absolute bg-black/90 backdrop-blur-xl border border-[#00ff88]/30 rounded-xl p-4 min-w-[220px] shadow-[0_0_30px_rgba(0,255,136,0.2)]"
          style={{
            left: `${((selectedStation.lng - 76.8) / 0.8) * 75 + 12}%`,
            top: `${((28.9 - selectedStation.lat) / 0.5) * 66 + 15}%`,
            transform: 'translate(-50%, -100%)'
          }}
        >
          <button 
            onClick={() => setSelectedStation(null)}
            className="absolute top-2 right-2 text-muted-foreground hover:text-foreground"
          >
            x
          </button>
          <h3 className="font-bold text-[#00ff88] mb-3">{selectedStation.name}</h3>
          <div className="space-y-2 text-sm">
            <div className="flex justify-between">
              <span className="text-muted-foreground">Load:</span>
              <span className="font-mono font-bold" style={{ color: getLoadColor(selectedStation.load) }}>
                {selectedStation.load}%
              </span>
            </div>
            <div className="flex justify-between">
              <span className="text-muted-foreground">Capacity:</span>
              <span className="text-foreground font-mono">{selectedStation.capacity} MW</span>
            </div>
            <div className="flex justify-between">
              <span className="text-muted-foreground">Status:</span>
              <span className={`uppercase text-xs font-bold px-2 py-0.5 rounded ${
                selectedStation.status === 'critical' ? 'bg-red-500/20 text-red-400' : 
                selectedStation.status === 'warning' ? 'bg-yellow-500/20 text-yellow-400' : 
                'bg-[#00ff88]/20 text-[#00ff88]'
              }`}>{selectedStation.status}</span>
            </div>
            <div className="mt-3 pt-2 border-t border-white/10">
              <div className="h-2 bg-black/40 rounded-full overflow-hidden">
                <div 
                  className="h-full rounded-full transition-all duration-500" 
                  style={{ width: `${selectedStation.load}%`, backgroundColor: getLoadColor(selectedStation.load) }} 
                />
              </div>
            </div>
          </div>
        </div>
      )}
    </div>
  )

  return (
    <div className="space-y-6 relative">
      {/* Background grid */}
      <div className="fixed inset-0 grid-pattern animate-grid-flow pointer-events-none opacity-30" />

      {/* Header */}
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4 relative z-10">
        <div>
          <h1 className="text-3xl font-bold text-foreground text-glow-green">Power Load Map</h1>
          <p className="text-muted-foreground mt-1">Real-time grid visualization across Delhi NCR</p>
        </div>
        <div className="flex items-center gap-3">
          <div className="flex items-center gap-2 px-3 py-1.5 rounded-full bg-[#00ff88]/10 border border-[#00ff88]/30">
            <span className="w-2 h-2 rounded-full bg-[#00ff88] shadow-[0_0_8px_#00ff88] animate-pulse" />
            <span className="text-xs font-semibold text-[#00ff88]">LIVE</span>
          </div>
          <Button 
            variant="outline" 
            size="sm" 
            onClick={handleRefresh}
            className="border-[#00d4ff]/30 text-[#00d4ff] hover:bg-[#00d4ff]/10"
          >
            <RefreshCw className={`w-4 h-4 mr-2 ${isRefreshing ? 'animate-spin' : ''}`} />
            Refresh
          </Button>
        </div>
      </div>

      {/* Stats Row */}
      <div className="grid grid-cols-2 lg:grid-cols-5 gap-4 relative z-10">
        <div className="glow-card rounded-xl p-4">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 rounded-xl bg-red-500/10 flex items-center justify-center">
              <AlertTriangle className="w-5 h-5 text-red-400" />
            </div>
            <div>
              <p className="text-[10px] text-muted-foreground uppercase tracking-wider">Critical</p>
              <p className="text-2xl font-bold text-red-400 font-mono">{criticalCount}</p>
            </div>
          </div>
        </div>
        <div className="glow-card rounded-xl p-4">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 rounded-xl bg-yellow-500/10 flex items-center justify-center">
              <Activity className="w-5 h-5 text-yellow-400" />
            </div>
            <div>
              <p className="text-[10px] text-muted-foreground uppercase tracking-wider">Warning</p>
              <p className="text-2xl font-bold text-yellow-400 font-mono">{warningCount}</p>
            </div>
          </div>
        </div>
        <div className="glow-card rounded-xl p-4">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 rounded-xl bg-[#00ff88]/10 flex items-center justify-center">
              <Zap className="w-5 h-5 text-[#00ff88]" />
            </div>
            <div>
              <p className="text-[10px] text-muted-foreground uppercase tracking-wider">Normal</p>
              <p className="text-2xl font-bold text-[#00ff88] font-mono">{normalCount}</p>
            </div>
          </div>
        </div>
        <div className="glow-card rounded-xl p-4">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 rounded-xl bg-[#00d4ff]/10 flex items-center justify-center">
              <Radio className="w-5 h-5 text-[#00d4ff]" />
            </div>
            <div>
              <p className="text-[10px] text-muted-foreground uppercase tracking-wider">Total Load</p>
              <p className="text-2xl font-bold text-[#00d4ff] font-mono">{Math.round(totalLoad)}<span className="text-sm ml-1">MW</span></p>
            </div>
          </div>
        </div>
        <div className="glow-card rounded-xl p-4 col-span-2 lg:col-span-1">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 rounded-xl bg-[#a855f7]/10 flex items-center justify-center">
              <Map className="w-5 h-5 text-[#a855f7]" />
            </div>
            <div>
              <p className="text-[10px] text-muted-foreground uppercase tracking-wider">Stations</p>
              <p className="text-2xl font-bold text-[#a855f7] font-mono">{gridStations.length}</p>
            </div>
          </div>
        </div>
      </div>

      {/* Map Container */}
      <div className="glow-card rounded-2xl overflow-hidden relative z-10">
        <div className="flex items-center justify-between p-4 border-b border-white/5">
          <div className="flex items-center gap-3">
            <div className="w-10 h-10 rounded-xl bg-[#00ff88]/10 flex items-center justify-center">
              <MapPin className="w-5 h-5 text-[#00ff88]" />
            </div>
            <div>
              <h3 className="text-lg font-bold text-foreground">Delhi NCR Power Grid Network</h3>
              <p className="text-xs text-muted-foreground">Last updated: {lastUpdate.toLocaleTimeString()}</p>
            </div>
          </div>
          <Button variant="ghost" size="sm" className="text-muted-foreground hover:text-foreground">
            <Layers className="w-4 h-4 mr-2" />
            Layers
          </Button>
        </div>
        
        <div className="relative h-[500px] lg:h-[600px]">
          {!isLoaded || loadError ? (
            <FallbackMap />
          ) : (
            <GoogleMap
              mapContainerStyle={mapContainerStyle}
              center={center}
              zoom={10}
              onLoad={onLoad}
              onUnmount={onUnmount}
              options={{
                styles: darkMapStyles,
                disableDefaultUI: true,
                zoomControl: false,
                mapTypeControl: false,
                streetViewControl: false,
                fullscreenControl: false,
              }}
            >
              {gridStations.map((station) => (
                <div key={station.id}>
                  <Circle
                    center={{ lat: station.lat, lng: station.lng }}
                    radius={station.load * 50}
                    options={{
                      fillColor: getLoadColor(station.load),
                      fillOpacity: 0.15,
                      strokeColor: getLoadColor(station.load),
                      strokeOpacity: 0.5,
                      strokeWeight: 2,
                    }}
                  />
                  <Marker
                    position={{ lat: station.lat, lng: station.lng }}
                    onClick={() => setSelectedStation(station)}
                    icon={{
                      path: google.maps.SymbolPath.CIRCLE,
                      fillColor: getStatusColor(station.status),
                      fillOpacity: 1,
                      strokeColor: "#ffffff",
                      strokeWeight: 2,
                      scale: 12,
                    }}
                  />
                </div>
              ))}
              {selectedStation && (
                <InfoWindow
                  position={{ lat: selectedStation.lat, lng: selectedStation.lng }}
                  onCloseClick={() => setSelectedStation(null)}
                >
                  <div className="p-2 min-w-[200px] bg-[#0a0a0f] text-white rounded-lg">
                    <h3 className="font-bold text-[#00ff88] mb-2">{selectedStation.name}</h3>
                    <div className="space-y-1 text-sm">
                      <div className="flex justify-between">
                        <span className="text-gray-400">Load:</span>
                        <span className="font-semibold" style={{ color: getLoadColor(selectedStation.load) }}>
                          {selectedStation.load}%
                        </span>
                      </div>
                      <div className="flex justify-between">
                        <span className="text-gray-400">Capacity:</span>
                        <span>{selectedStation.capacity} MW</span>
                      </div>
                      <div className="mt-2 pt-2 border-t border-gray-700">
                        <div className="w-full bg-gray-700 rounded-full h-2">
                          <div 
                            className="h-2 rounded-full" 
                            style={{ width: `${selectedStation.load}%`, backgroundColor: getLoadColor(selectedStation.load) }} 
                          />
                        </div>
                      </div>
                    </div>
                  </div>
                </InfoWindow>
              )}
            </GoogleMap>
          )}

          {/* Map Controls */}
          <div className="absolute top-4 right-4 flex flex-col gap-2">
            <Button size="icon" onClick={handleZoomIn} className="bg-black/80 backdrop-blur-sm border border-white/10 hover:bg-black/90 text-foreground">
              <ZoomIn className="w-4 h-4" />
            </Button>
            <Button size="icon" onClick={handleZoomOut} className="bg-black/80 backdrop-blur-sm border border-white/10 hover:bg-black/90 text-foreground">
              <ZoomOut className="w-4 h-4" />
            </Button>
            <Button size="icon" onClick={handleCenter} className="bg-black/80 backdrop-blur-sm border border-white/10 hover:bg-black/90 text-foreground">
              <Locate className="w-4 h-4" />
            </Button>
          </div>

          {/* Legend */}
          <div className="absolute bottom-4 left-4 bg-black/80 backdrop-blur-xl border border-white/10 rounded-xl p-4">
            <p className="text-xs font-bold text-foreground mb-3 uppercase tracking-wider">Load Status</p>
            <div className="space-y-2">
              {[
                { color: "#ff4444", label: "Critical", range: ">85%" },
                { color: "#ffaa00", label: "Warning", range: "70-85%" },
                { color: "#00ff88", label: "Normal", range: "50-70%" },
                { color: "#00d4ff", label: "Low", range: "<50%" },
              ].map((item) => (
                <div key={item.label} className="flex items-center gap-3">
                  <span className="w-3 h-3 rounded-full shadow-[0_0_8px_currentColor]" style={{ backgroundColor: item.color, color: item.color }} />
                  <span className="text-xs text-muted-foreground">{item.label}</span>
                  <span className="text-[10px] text-muted-foreground/60 ml-auto">{item.range}</span>
                </div>
              ))}
            </div>
          </div>
        </div>
      </div>

      {/* Station Cards */}
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4 relative z-10">
        {gridStations.slice(0, 8).map((station) => (
          <div 
            key={station.id}
            className="glow-card rounded-xl p-4 cursor-pointer hover:scale-[1.02] transition-transform"
            onClick={() => setSelectedStation(station)}
          >
            <div className="flex items-start justify-between mb-3">
              <div className="flex-1 min-w-0">
                <p className="text-sm font-semibold text-foreground truncate">{station.name}</p>
                <p className="text-[10px] text-muted-foreground uppercase tracking-wider">{station.type}</p>
              </div>
              <span className={`px-2 py-0.5 rounded-full text-[10px] uppercase font-bold ${
                station.status === 'critical' ? 'bg-red-500/20 text-red-400' : 
                station.status === 'warning' ? 'bg-yellow-500/20 text-yellow-400' : 
                'bg-[#00ff88]/20 text-[#00ff88]'
              }`}>{station.status}</span>
            </div>
            <div className="flex items-end justify-between">
              <div>
                <span className="text-3xl font-bold font-mono" style={{ color: getLoadColor(station.load) }}>
                  {station.load}
                </span>
                <span className="text-sm text-muted-foreground ml-1">%</span>
              </div>
              <div className="text-right">
                <p className="text-xs text-muted-foreground">{station.capacity} MW</p>
              </div>
            </div>
            <div className="mt-3 h-1.5 bg-black/40 rounded-full overflow-hidden">
              <div 
                className="h-full rounded-full transition-all duration-500" 
                style={{ width: `${station.load}%`, backgroundColor: getLoadColor(station.load), boxShadow: `0 0 8px ${getLoadColor(station.load)}` }} 
              />
            </div>
          </div>
        ))}
      </div>

      {/* Footer */}
      <div className="text-center py-4 border-t border-border/50 relative z-10">
        <p className="text-xs text-muted-foreground">
          Smart AI Grids - Real-time Grid Monitoring | Team Power Rangers
        </p>
      </div>
    </div>
  )
}
