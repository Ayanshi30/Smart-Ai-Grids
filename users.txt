const getRoleColor = (role: string) => {
  switch (role) {
    case "Admin": return "bg-[#a855f7]/10 text-[#a855f7] border-[#a855f7]/30"
    case "Operator": return "bg-[#00ff88]/10 text-[#00ff88] border-[#00ff88]/30"
    case "Analyst": return "bg-[#00d4ff]/10 text-[#00d4ff] border-[#00d4ff]/30"
    case "Technician": return "bg-[#FFD700]/10 text-[#FFD700] border-[#FFD700]/30"
    default: return "bg-muted text-muted-foreground border-border"
  }
}
